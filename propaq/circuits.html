 70        """
 71        Construct a MajoranaCircuit from a Qiskit QuantumCircuit.
 72
 73        ### TODO: 
 74        Currently, only a subset of Qiskit gates are supported. Supported gates 
 75        include those that arise in the Local Unitary Cluster Jastrow (LUCJ) ansatz. 
 76        However, we hope to extend this to a more general set of gates in the future.
 77
 78        Arguments:
 79            qc: A Qiskit QuantumCircuit to convert.
 80            n_modes: The number of Majorana modes in the system.
 81
 82        Returns:
 83            A MajoranaCircuit initialized with the given Qiskit circuit.
 84        """
 85        def _mark_intermediate(rots: list[MajoranaRotation]) -> list[MajoranaRotation]:
 86            for i, rot in enumerate(rots):
 87                rot.is_intermediate = i < len(rots) - 1
 88            return rots
 89
 90        all_layers: list[list[MajoranaRotation]] = []
 91        qiskit_gate_idx: int = 0
 92
 93        for layer in circuit_to_dag(qc).layers():
 94            layer_rots: list[MajoranaRotation] = []
 95            for node in layer["graph"].topological_op_nodes():
 96                instr = node.op
 97                qargs = node.qargs
 98
 99                if instr.name in ["measure", "barrier"]:
100                    continue
101                if instr.name not in ["xx_plus_yy", "p", "rz", "cp", "x", "swap"]:
102                    raise ValueError(
103                        f"Unsupported gate {instr.name} in Qiskit circuit. "
104                        "Supported gates: xx_plus_yy, p, rz, cp, x, swap."
105                    )
106
107                q_indices = [qc.find_bit(q).index for q in qargs]
108
109                if instr.name == "xx_plus_yy":
110                    if len(qargs) != 2:
111                        raise ValueError("xx_plus_yy gate must have exactly 2 qubits.")
112                    beta = float(instr.params[1]) if len(instr.params) > 1 else 0.0
113
114                    rots: list[MajoranaRotation] = []
115                    if abs(beta) > 1e-14:
116                        rz_sum = MajoranaTermSum[MajoranaMonomial].from_rz_angle(
117                            q_indices[1], -beta, n_modes
118                        )
119                        for gen, ang in rz_sum.items():
120                            rots.append(MajoranaRotation(gen, float(ang.real)))
121
122                    majoranasum: MajoranaTermSum[MajoranaMonomial] = (
123                        MajoranaTermSum[MajoranaMonomial].from_xx_plus_yy(instr, q_indices, n_modes)
124                    )
125                    for gen, ang in majoranasum.items():
126                        rots.append(MajoranaRotation(gen, float(ang.real)))
127
128                    if abs(beta) > 1e-14:
129                        rz_neg_sum = MajoranaTermSum[MajoranaMonomial].from_rz_angle(
130                            q_indices[1], beta, n_modes
131                        )
132                        for gen, ang in rz_neg_sum.items():
133                            rots.append(MajoranaRotation(gen, float(ang.real)))
134
135                    _mark_intermediate(rots)
136                    for rot in rots:
137                        rot.qiskit_gate_idx = qiskit_gate_idx
138                    layer_rots.extend(rots)
139                    qiskit_gate_idx += 1
140                    continue
141
142                elif instr.name == "p":
143                    majoranasum = MajoranaTermSum[MajoranaMonomial].from_phase(instr, q_indices, n_modes)
144
145                elif instr.name == "rz":
146                    majoranasum = MajoranaTermSum[MajoranaMonomial].from_rz(instr, q_indices, n_modes)
147
148                elif instr.name == "cp":
149                    if len(qargs) != 2:
150                        raise ValueError("cp gate must have exactly 2 qubits.")
151                    majoranasum = MajoranaTermSum[MajoranaMonomial].from_cp(instr, q_indices, n_modes)
152
153                elif instr.name == "swap":
154                    if len(qargs) != 2:
155                        raise ValueError("swap gate must have exactly 2 qubits.")
156                    majoranasum = MajoranaTermSum[MajoranaMonomial].from_swap(instr, q_indices, n_modes)
157
158                elif instr.name == "x":
159                    if len(qargs) != 1:
160                        raise ValueError("x gate must have exactly 1 qubit.")
161                    majoranasum = MajoranaTermSum[MajoranaMonomial].from_x(instr, q_indices, n_modes)
162
163                items = list(majoranasum.items())
164                rots = [MajoranaRotation(gen, float(ang.real)) for gen, ang in items]
165                _mark_intermediate(rots)
166                for rot in rots:
167                    rot.qiskit_gate_idx = qiskit_gate_idx
168                layer_rots.extend(rots)
169                qiskit_gate_idx += 1
170
171            if layer_rots:
172                all_layers.append(layer_rots)
173
174        mc = cls.__new__(cls)
175        mc._layers = all_layers
176        mc.n_modes = n_modes
177        return mc
