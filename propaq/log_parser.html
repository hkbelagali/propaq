147class LogParser:
148    """
149    Parse a propaq JSONL log file into typed event lists.
150    """
151
152    def __init__(self, filename: str) -> None:
153        """Build a LogParser for the given log file, reading all events into memory."""
154        self._filename = filename
155        self._gate_events: list[GateEvent] = []
156        self._truncation_events: list[TruncationEvent] = []
157        self._surrogate_flush_events: list[SurrogateFlushEvent] = []
158        self._surrogate_flush_deferred_events: list[SurrogateFlushDeferredEvent] = []
159        self._surrogate_merge_events: list[SurrogateMergeEvent] = []
160        self._load()
161
162    def _load(self) -> None:
163        self._gate_events.clear()
164        self._truncation_events.clear()
165        self._surrogate_flush_events.clear()
166        self._surrogate_flush_deferred_events.clear()
167        self._surrogate_merge_events.clear()
168        with open(self._filename) as f:
169            for line in f:
170                line = line.strip()
171                if not line:
172                    continue
173                ev = json.loads(line)
174                kind = ev["event"]
175                if kind == "gate":
176                    self._gate_events.append(GateEvent(
177                        gate_idx=ev["gate_idx"],
178                        layer_idx=ev["layer_idx"],
179                        map_terms=ev["map_terms"],
180                        outbox_terms=ev["outbox_terms"],
181                        avg_ms_per_gate=ev.get("avg_ms_per_gate"),
182                        qiskit_gate_idx=ev.get("qiskit_gate_idx"),
183                        monomials=ev.get("monomials"),
184                    ))
185                elif kind == "truncation":
186                    self._truncation_events.append(TruncationEvent(
187                        gate_idx=ev["gate_idx"],
188                        layer_idx=ev["layer_idx"],
189                        trigger=ev["trigger"],
190                        terms_before=ev["terms_before"],
191                        terms_after=ev["terms_after"],
192                        terms_discarded=ev["terms_discarded"],
193                        discarded_coeff_l1=ev["discarded_coeff_l1"],
194                        discarded_coeff_max=ev["discarded_coeff_max"],
195                        weight_cutoff=ev["weight_cutoff"],
196                        coeff_cutoff=ev["coeff_cutoff"],
197                        elapsed_ms=ev["elapsed_ms"],
198                        qiskit_gate_idx=ev.get("qiskit_gate_idx"),
199                    ))
200                elif kind == "surrogate_flush":
201                    self._surrogate_flush_events.append(SurrogateFlushEvent(
202                        gate_idx=ev["gate_idx"],
203                        layer_idx=ev["layer_idx"],
204                        trigger=ev["trigger"],
205                        terms_before=ev["terms_before"],
206                        terms_after=ev["terms_after"],
207                        terms_discarded=ev["terms_discarded"],
208                        monomials_before=ev["monomials_before"],
209                        monomials_after=ev["monomials_after"],
210                        monomials_discarded=ev["monomials_discarded"],
211                        frequency=ev["frequency"],
212                        weight=ev["weight"],
213                        coefficient=ev["coefficient"],
214                        elapsed_ms=ev["elapsed_ms"],
215                        qiskit_gate_idx=ev.get("qiskit_gate_idx"),
216                    ))
217                elif kind == "surrogate_flush_deferred":
218                    self._surrogate_flush_deferred_events.append(SurrogateFlushDeferredEvent(
219                        gate_idx=ev["gate_idx"],
220                        layer_idx=ev["layer_idx"],
221                        trigger=ev["trigger"],
222                        terms=ev["terms"],
223                        monomials=ev["monomials"],
224                        reason=ev["reason"],
225                        qiskit_gate_idx=ev.get("qiskit_gate_idx"),
226                    ))
227                elif kind == "surrogate_merge":
228                    self._surrogate_merge_events.append(SurrogateMergeEvent(
229                        gate_idx=ev["gate_idx"],
230                        layer_idx=ev["layer_idx"],
231                        terms_before=ev["terms_before"],
232                        terms_after=ev["terms_after"],
233                        monomials_after=ev["monomials_after"],
234                        qiskit_gate_idx=ev.get("qiskit_gate_idx"),
235                    ))
236
237    def reload(self) -> None:
238        """Re-read the log file, picking up any new events appended since construction."""
239        self._load()

243        """All gate events in file order."""

260

263        """All merge-only cadence events in file order (surrogate propagators only)."""

266    @property
267    def gate_indices(self) -> list[int]:
268        """Indices of gates at which events were logged."""

272    def map_terms(self) -> list[int]:
273        """Terms in local hashmaps, sampled at each logged gate."""
274        return [e.map_terms for e in self._gate_events]

276    @property
277    def outbox_terms(self) -> list[int]:
278        """Terms in outboxes (may include duplicates), sampled at each logged gate."""
279        return [e.outbox_terms for e in self._gate_events]
280
281    @property
282    def monomials(self) -> list[int | None]:
283        """Live monomial count sampled at each logged gate (surrogate propagators
284        only; None per entry for numerical propagators or old log files)."""
285        return [e.monomials for e in self._gate_events]
286
287    @property
288    def terms_before(self) -> list[int]:
289        """Deduplicated term count before each truncation (numerical propagators only)."""
290        return [e.terms_before for e in self._truncation_events]
291
292    @property
293    def terms_after(self) -> list[int]:
294        """Term count after each truncation (numerical propagators only)."""
295        return [e.terms_after for e in self._truncation_events]
296
297    @property
298    def terms_discarded(self) -> list[int]:
299        """Number of terms dropped at each truncation (numerical propagators only)."""
300        return [e.terms_discarded for e in self._truncation_events]
301
302    @property
303    def discarded_coeff_l1(self) -> list[float]:
304        """Sum of |coeff| discarded at each truncation (spectral weight lost)."""
305        return [e.discarded_coeff_l1 for e in self._truncation_events]
306
307    @property
308    def discarded_coeff_max(self) -> list[float]:
309        """Largest |coeff| discarded at each truncation (worst-case information loss)."""
310        return [e.discarded_coeff_max for e in self._truncation_events]
311
312    @property
313    def qiskit_gate_indices(self) -> list[int | None]:
314        """Qiskit gate index at each logged gate event, or None for non-Qiskit circuits."""
315        return [e.qiskit_gate_idx for e in self._gate_events]
316
317    @property
318    def avg_ms_per_gate(self) -> list[float | None]:
319        """Average ms/gate between consecutive gate log events (None for the first event)."""
320        return [e.avg_ms_per_gate for e in self._gate_events]
321
322    @property
323    def elapsed_ms(self) -> list[float]:
324        """Wall time (ms) for each flush+truncation step (numerical propagators only)."""
325        return [e.elapsed_ms for e in self._truncation_events]
326
327    @property
328    def monomials_before(self) -> list[int]:
329        """Exact total monomial count before each surrogate flush (surrogate propagators only)."""
330        return [e.monomials_before for e in self._surrogate_flush_events]
331
332    @property
333    def monomials_after(self) -> list[int]:
334        """Exact total monomial count after each surrogate flush (surrogate propagators only)."""
335        return [e.monomials_after for e in self._surrogate_flush_events]
336
337    @property
338    def monomials_discarded(self) -> list[int]:
339        """Number of monomials dropped at each surrogate flush (surrogate propagators only)."""
340        return [e.monomials_discarded for e in self._surrogate_flush_events]
